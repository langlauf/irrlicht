00231         tmp.W = (other.W * W) - (other.X * X) - (other.Y * Y) - (other.Z * Z);

00396 inline quaternion& quaternion::slerp(quaternion q1, quaternion q2, f32 time)
00397 {
00398         f32 angle = q1.dotProduct(q2);
00399 
00400         if (angle < 0.0f)
00401         {
00402                 q1 *= -1.0f;
00403                 angle *= -1.0f;
00404         }
00405 
00406         f32 scale;
00407         f32 invscale;
00408 
00409         if ((angle + 1.0f) > 0.05f)
00410         {
00411                 if ((1.0f - angle) >= 0.05f) // spherical interpolation
00412                 {
00413                         const f32 theta = acosf(angle);
00414                         const f32 invsintheta = reciprocal(sinf(theta));
00415                         scale = sinf(theta * (1.0f-time)) * invsintheta;
00416                         invscale = sinf(theta * time) * invsintheta;
00417                 }
00418                 else // linear interploation
00419                 {
00420                         scale = 1.0f - time;
00421                         invscale = time;
00422                 }
00423         }
00424         else
00425         {
00426                 q2.set(-q1.Y, q1.X, -q1.W, q1.Z);
00427                 scale = sinf(PI * (0.5f - time));
00428                 invscale = sinf(PI * time);
00429         }
00430 
00431         return (*this = (q1*scale) + (q2*invscale));

00520 {

00522         // Copy, since cannot modify local

00524         vector3df v1 = to;
00525         v0.normalize();
00526         v1.normalize();
00527 
00528         const f32 d = v0.dotProduct(v1);
00529         if (d >= 1.0f) // If dot == 1, vectors are the same
00530         {
00531                 return makeIdentity();

00535         const f32 invs = 1.f / s;

00538         X = c.X;
00539         Y = c.Y;
00540         Z = c.Z;
00541         W = s * 0.5f;
